1.Создай таблицу books с полями: id, title VARCHAR(100) NOT NULL,
 author VARCHAR(100), genreVARCHAR(50), price NUMERIC(5,2), pages INT, year INT.
CREATE TABLE books(
id SERIAL PRIMARY KEY,
title VARCHAR(100) NOT NULL,
author VARCHAR(100),
genre VARCHAR(50),
price NUMERIC(5,2),
pages INT,
year INT
);
